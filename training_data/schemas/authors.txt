cube Paper
  dim  Paper.Id (number) -- Id of the paper
  dim  Paper.Title (string) -- Title of the paper
  dim  Paper.Year (number) -- Year of the paper commonsense reasoning: if the year is "0", it means this paper is preprint, or not published
  dim  Paper.ConferenceId (number) -- Conference Id in which paper was published
  dim  Paper.JournalId (number) -- Journal Id in which paper was published commonsense reasoning: If a paper contain "0" in both ConferenceID and JournalId, it means this paper is preprint
  dim  Paper.Keyword (string) -- Keywords of the paper commonsense reasoning: Keywords should contain words and phrases that suggest what the topic is about. Similar keywords represent similar fields or sub-field.
  meas Paper.count (count)
  meas Paper.sum_Year (sum)
  meas Paper.avg_Year (avg)
  meas Paper.min_Year (min)
  meas Paper.max_Year (max)
