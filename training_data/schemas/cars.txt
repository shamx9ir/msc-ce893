cube data
  dim  data.ID (number) -- unique ID for each car
  dim  data.mpg (number) -- mileage of the car in miles per gallon The car with higher mileage is more fuel-efficient.
  dim  data.cylinders (number) -- the number of cylinders present in the car
  dim  data.displacement (number) -- engine displacement in cubic mm sweep volume = displacement / no_of cylinders
  dim  data.horsepower (number) -- horse power associated with the car horse power is the metric used to indicate the power produced by a car's engine - the higher the number, the more power is sent to the wheels and, in theory, the faster it will go.
  dim  data.weight (number) -- weight of the car in lbs A bigger, heavier vehicle provides better crash protection than a smaller
  dim  data.acceleration (number) -- acceleration of the car in miles per squared hour
  dim  data.model (number) -- the year when the car model was introduced in the market 0 --> 1970
  dim  data.car_name (string) -- name of the car
  meas data.count (count)
  meas data.sum_mpg (sum)
  meas data.avg_mpg (avg)
  meas data.min_mpg (min)
  meas data.max_mpg (max)
  meas data.sum_cylinders (sum)

  meas data.max_displacement (max)

  meas data.max_horsepower (max)
  meas data.sum_weight (sum)
