cube Crime
  dim  Crime.report_no (number) -- unique Numeric identifier of the crime report.
  dim  Crime.case_number (string) -- Case number assigned to the reported crime. There is usually one case number for any reported crime except in the case of multiple homicide where a single report will produce one case number for each victim if case_number appears > 1, it means this is a multiple homicide
  dim  Crime.date (string) -- Date of the occurrence of the incident being reported
  dim  Crime.block (string) -- A redacted address indicating in which city block the incident occurred
  dim  Crime.iucr_no (string) -- Illinois Uniform Crime Reporting code: a four digit code used to classify criminal incidents when taking reports.
  dim  Crime.location_description (string) -- A description of the kind of area where incident occurred
  dim  Crime.arrest (string) -- A true/false value indicating if an arrest has been made
  dim  Crime.domestic (string) -- A true/false value indicating if the incident is as case of domestic violence
  dim  Crime.beat (number)
  dim  Crime.district_no (number) -- A number indicating the geographic subdivision of the police district where the incident occurred
  dim  Crime.ward_no (number) -- A two digit number identifying the legislative district (ward) where the incident occurred.
  dim  Crime.community_area_no (number) -- A number identifying one of 77 geographic areas where the incident occurred
  dim  Crime.fbi_code_no (string) -- A code identifying the type of crime reported according to the classification scheme used by the FBI.
  dim  Crime.latitude (string) -- The latitude where incident is reported to have occurred.
  dim  Crime.longitude (string) -- The longitude where incident is reported to have occurred. The precise location / coordinate: combines the longitude and latitude for plotting purposes. (latitude, longitude)
  meas Crime.count (count)
  meas Crime.sum_beat (sum)
  meas Crime.avg_beat (avg)
  meas Crime.min_beat (min)
  meas Crime.max_beat (max)
