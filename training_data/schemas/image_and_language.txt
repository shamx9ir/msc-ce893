cube PRED_CLASSES
  dim  PRED_CLASSES.PRED_CLASS_ID (number) -- the unique prediction id for the class

cube IMG_REL
  dim  IMG_REL.row_key (string)
  dim  IMG_REL.IMG_ID (number) -- the image id
  dim  IMG_REL.PRED_CLASS_ID (number) -- the prediction relationship class between objects if OBJ1_SAMPLE_ID = OBJ2_SAMPLE_ID, this relationship is the self-relation.
  dim  IMG_REL.OBJ1_SAMPLE_ID (number) -- the sample id of the first object
  dim  IMG_REL.OBJ2_SAMPLE_ID (number) -- the sample id of the second object if (OBJ1_SAMPLE_ID, OBJ2_SAMPLE_ID) has multiple PRED_CLASS_ID, it means these two objects have multiple relations.
  meas IMG_REL.count (count)
