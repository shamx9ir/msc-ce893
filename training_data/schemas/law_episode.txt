cube Episode
  dim  Episode.episode_id (string) -- the unique identifier of the episode
  dim  Episode.series (string) -- the name of the series
  dim  Episode.season (number) -- a number indicating the season of the episode
  dim  Episode.episode (number) -- the sequential number of the episode within a specific season
  dim  Episode.number_in_series (number) -- the overall sequence number of episode in the entire series

  dim  Episode.summary (string) -- a brief description of what happens in the episode
  dim  Episode.air_date (time) -- the date the episode was initially broadcast in the United States
  dim  Episode.episode_image (string) -- a link to an image from the episode

  dim  Person.name (string) -- the name of the person
  dim  Person.birthdate (time) -- the date the person was born if null, it means this birthdate of the person is not available
  dim  Person.birth_name (string) -- the name parents gave to the person shortly after birth
  dim  Person.birth_place (string) -- the place where a person was born. the birth place info is integrate if birth_name, birth_place, birth_region don't contain null value the full birth place = birth_country +birth_region +birth_place
  dim  Person.birth_region (string) -- the geographical area describing where a person was born that is larger than the birth place
  dim  Person.birth_country (string) -- the name of the country where the person was born can ask questions about its corresponding continent: e.g.: USA --> North America
  dim  Person.height_meters (number) -- how tall the person is in meters
  dim  Person.nickname (string) -- the nickname the person goes by
  meas Person.count (count)

cube Credit
  dim  Credit.row_key (string)
  dim  Credit.episode_id (string) -- the id of the episode to which the credit information pertains
  dim  Credit.person_id (string) -- the id of the person to which the credit information pertains
  dim  Credit.category (string) -- the kind of credit being recognized
  dim  Credit.role (string) -- the role for which the person is being recognized If the credit is for an actor, this is the name of the character he or she portrayed (Dan Florek plays Donald Craven). Otherwise, this is the production role (producer, sound editor).
  dim  Credit.credited (string) -- whether the credit was displayed in the credits at the end of the episode A person may fill a role in an episode, but not show up in the on-screen credits. In this case, the work is said to be "uncredited."
  meas Credit.count (count)
