cube Dish
  dim  Dish.id (number) -- unique id number indicating the dishes
  dim  Dish.name (string) -- the name of the dish
  dim  Dish.description (string) -- description of the dish (no value)
  dim  Dish.menus_appeared (number) -- how many menus have this dish
  dim  Dish.times_appeared (number) -- how many times this dish appears in the menu if times_appeared > menus_appeard: this dish appeared in a menu more than once
  dim  Dish.first_appeared (number) -- the year that this dish appears first 1. the year outside of [1851, 2012], it means this data is not correct 2. if this dish lasts longer (last_appeared - first_appeard), it means its history is long or historical / classical
  dim  Dish.last_appeared (number) -- the year that this dish appears the last time 1. the year outside of [1851, 2012], it means this data is not correct 2. if this dish lasts longer (last_appeared - first_appeard), it means its history is long or historical / classical
  dim  Dish.lowest_price (number) -- the lowest price of the dish 0: for free
  dim  Dish.highest_price (number) -- the highest price of the dish
  meas Dish.count (count)
  meas Dish.sum_menus_appeared (sum)
  meas Dish.avg_menus_appeared (avg)
  meas Dish.min_menus_appeared (min)
  meas Dish.max_menus_appeared (max)

cube Menu
  dim  Menu.id (number) -- the unique number identifying the menu
  dim  Menu.name (string) -- the name of the menu if the value is not null or empty, it means this menu has special dishes. otherwise, this menu is general and nothing special
  dim  Menu.sponsor (string) -- the sponsor of this menu if the value is null or empyt, it means this meanu is DIY by the restaurant.
  dim  Menu.event (string) -- the event that the menu was created for
  dim  Menu.venue (string) -- the venue that the menu was created for
  dim  Menu.place (string) -- the location that the menu was used the location that the menu was used
  dim  Menu.physical_description (string) -- physical description of the menu
  dim  Menu.occasion (string) -- occasion of the menu
  dim  Menu.notes (string) -- notes
  dim  Menu.call_number (string) -- call number if null: not support for taking out or booking in advance

  dim  Menu.language (string) -- language not useful
  dim  Menu.date (time) -- the date that this menu was created
  dim  Menu.location (string) -- the location that the menu was used
