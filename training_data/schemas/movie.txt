  dim  actor.Name (string) -- actor's name

  dim  actor.Birth_City (string) -- actor's birth city
  dim  actor.Birth_Country (string) -- actor's birth country
  dim  actor.Height_Inches (number) -- actor's height the unit is inch
  dim  actor.Biography (string) -- actor's biography
  dim  actor.Gender (string) -- actor's gender
  dim  actor.Ethnicity (string) -- actor's ethnicity
  dim  actor.NetWorth (string) -- actor's networth The actor with more networth is richer.

  meas actor.sum_Height_Inches (sum)
  meas actor.avg_Height_Inches (avg)
  meas actor.min_Height_Inches (min)
  meas actor.max_Height_Inches (max)
