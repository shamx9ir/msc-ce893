cube users
  dim  users.userid (number) -- unique identifier number of users
  dim  users.age (string) -- age 1: 1-18 years old; 18: 18-25 years old; 25: 25-35 years old; 35: 35-45 years old; 45: 45-50 years old; 50: 50-56 years old; 56: over 56 years old
  dim  users.u_gender (string) -- user gender M / F: Male / Female
  dim  users.occupation (string) -- occupation
  meas users.count (count)
