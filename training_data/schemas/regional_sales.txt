cube Sales_Orders
  dim  Sales_Orders.OrderNumber (string) -- the unique number identifying the order
  dim  Sales_Orders.Sales_Channel (string) -- Sales Channel  In-Store  Online  Distributor  Wholesale
  dim  Sales_Orders.WarehouseCode (string) -- Warehouse Code if the warehouse code is the same, it means this product is shipped from the same ware house
  dim  Sales_Orders.ProcuredDate (string) -- Procured Date date: month/date/year, 17--> 2017
  dim  Sales_Orders.OrderDate (string) -- Order Date
  dim  Sales_Orders.ShipDate (string) -- Ship Date
  dim  Sales_Orders.DeliveryDate (string) -- Delivery Date the difference "DeliveryDate - OrderDate" is smaller, it means faster delivery
  dim  Sales_Orders.CurrencyCode (string) -- Currency Code USD
  dim  Sales_Orders.SalesTeamID (number) -- Sales Team ID
  dim  Sales_Orders.CustomerID (number) -- _Customer ID
  dim  Sales_Orders.StoreID (number) -- _Store ID
  dim  Sales_Orders.ProductID (number) -- _Product ID
  dim  Sales_Orders.Order_Quantity (number) -- Order Quantity 1 - 8: higher means order quantity is higher or customer needs more
  dim  Sales_Orders.Discount_Applied (number) -- Discount Applied 0.2: 20% discount
  dim  Sales_Orders.Unit_Price (string)
  dim  Sales_Orders.Unit_Cost (string) -- net profit = Unit Price - Unit Cost
  meas Sales_Orders.count (count)
  meas Sales_Orders.sum_Order_Quantity (sum)
  meas Sales_Orders.avg_Order_Quantity (avg)
  meas Sales_Orders.min_Order_Quantity (min)
  meas Sales_Orders.max_Order_Quantity (max)
  meas Sales_Orders.sum_Discount_Applied (sum)
