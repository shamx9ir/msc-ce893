cube chapters
  dim  chapters.id (number) -- unique id number identifying the chapter
  dim  chapters.Act (number) -- An act is a major division of a theatre work, including a play, film, opera, or musical theatre An act can consist of one or more scenes
  dim  chapters.Scene (number) -- A scene is a dramatic part of a story, at a specific time and place, between specific characters.
  dim  chapters.Description (string) -- textual description of the chapter.
  dim  chapters.work_id (number) -- id number identifying the work
  meas chapters.count (count)
  meas chapters.sum_Act (sum)
  meas chapters.avg_Act (avg)
  meas chapters.min_Act (min)
  meas chapters.max_Act (max)
  meas chapters.sum_Scene (sum)

  meas chapters.max_Scene (max)
