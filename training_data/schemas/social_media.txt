  dim  location.Country (string) -- the country
  dim  location.State (string) -- the state

  dim  location.City (string) -- the city

cube twitter
  dim  twitter.TweetID (string) -- the unique id of the tweet
  dim  twitter.Weekday (string) -- the weekday that the tweet has been posted
  dim  twitter.Hour (number) -- the hour that the tweet has been posted
  dim  twitter.Day (number) -- the day that the tweet has been posted
  dim  twitter.Lang (string) -- the language of the tweet
  dim  twitter.IsReshare (string) -- whether the tweet is reshared A reshared tweet is when someone reshares or forwards a post to their own Twitter  true: the tweet is reshared  false: the tweet isn't reshared
  dim  twitter.Reach (number) -- the reach of the tweet reach counts the number of unique users who have seen your tweet
  dim  twitter.RetweetCount (number) -- the retweet count
  dim  twitter.Likes (number) -- the number of likes
  dim  twitter.Klout (number) -- the klout of the tweet
  dim  twitter.Sentiment (number) -- the sentiment of the tweet  >0: the sentiment of the tweet is positive.  =0: the sentiment of the tweet is neutral.  <0: the sentiment of the tweet is negative.
  dim  twitter.text (string) -- the text of the tweet
  dim  twitter.LocationID (number) -- the location id of the poster
  dim  twitter.UserID (string) -- the user id of the poster
  meas twitter.count (count)
