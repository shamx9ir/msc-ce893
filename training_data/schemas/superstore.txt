cube people
  dim  people.row_key (string)
  dim  people.Customer_ID (string) -- the id of the customers
  dim  people.Customer_Name (string) -- the name of the customers
  dim  people.Segment (string) -- the segment that the customers belong to  consumer  home office: synonym: headquarter.  corporate
  dim  people.Country (string) -- the country of people
  dim  people.City (string) -- the city of people
  dim  people.State (string) -- the state of people please mention its full name in the question, by referring to https://www23.statcan.gc.ca/imdb/p3VD.pl?Function=getVD&TVD=53971 e.g., New York --> NY
  dim  people.Postal_Code (number) -- the postal code
  dim  people.Region (string) -- the region of people  Central:  East:  West:  South:
  meas people.count (count)
  meas people.sum_Postal_Code (sum)
  meas people.avg_Postal_Code (avg)
  meas people.min_Postal_Code (min)
  meas people.max_Postal_Code (max)

cube central_superstore
  dim  central_superstore.Row_ID (number) -- the unique id for rows
  dim  central_superstore.Order_ID (string) -- the unique identifier for the order
  dim  central_superstore.Order_Date (time) -- the date of the order yyyy-mm-dd
  dim  central_superstore.Ship_Date (time) -- the date of the shipment yyyy-mm-dd 'shipment time' refers to the time interval between order_date and ship_date.
  dim  central_superstore.Ship_Mode (string) -- the ship mode of the order First Class / Second Class / Standard Class Among three ship modes, First Class has the fastest delivery speed, followed by Second Class and the speed of the Standard Class is the slowest.
  dim  central_superstore.Customer_ID (string) -- the id of the customer
  dim  central_superstore.Region (string) -- region of the customer's address
  dim  central_superstore.Product_ID (string) -- the id of the product
  dim  central_superstore.Sales (number) -- the sales of the product
  dim  central_superstore.Quantity (number) -- the quantity of the product
  dim  central_superstore.Discount (number) -- the discount of the product original price = sales / (1- discount)
  dim  central_superstore.Profit (number) -- the profit that the company got by selling the product total cost of products = sales / (1- discount) * quantity - profit deficiency: if the value is negative
  meas central_superstore.count (count)
  meas central_superstore.sum_Sales (sum)
  meas central_superstore.avg_Sales (avg)
  meas central_superstore.min_Sales (min)
  meas central_superstore.max_Sales (max)
  meas central_superstore.sum_Quantity (sum)
  meas central_superstore.avg_Quantity (avg)
  meas central_superstore.min_Quantity (min)
  meas central_superstore.max_Quantity (max)
  meas central_superstore.sum_Discount (sum)
  meas central_superstore.avg_Discount (avg)
  meas central_superstore.min_Discount (min)
  meas central_superstore.max_Discount (max)
  meas central_superstore.sum_Profit (sum)
  meas central_superstore.avg_Profit (avg)
  meas central_superstore.min_Profit (min)
  meas central_superstore.max_Profit (max)
