cube patients
  dim  patients.patient (string) -- the unique id for the patient
  dim  patients.birthdate (time) -- the birth date of the patient
  dim  patients.deathdate (time) -- the death date of the patient  the age of the patient = death year - birth year  if null, it means this patient is still alive
  dim  patients.ssn (string) -- the social security number of the patient
  dim  patients.drivers (string) -- the driver number of the patient if not, this patient doesn't have driving license
  dim  patients.passport (string) -- the passport number if not, this patient cannot go abroad, vice versa
  dim  patients.prefix (string) -- the prefix
  dim  patients.first (string) -- the first name
  dim  patients.last (string) -- the last name full name = first + last
  dim  patients.suffix (string) -- the suffix of the patient if suffix = PhD, JD, MD, it means this patient has doctoral degree. Otherwise, this patient is not.
  dim  patients.maiden (string) -- the maiden name of the patient Only married women have the maiden name
  dim  patients.marital (string) -- the marital status of the patient  M: married  S: single
  dim  patients.race (string) -- the race of the patient
  dim  patients.ethnicity (string) -- the ethnicity of the patient
  dim  patients.gender (string) -- the gender of the patient
  dim  patients.birthplace (string) -- the birth place
  dim  patients.address (string) -- the specific address
  meas patients.count (count)
